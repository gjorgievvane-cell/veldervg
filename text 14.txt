    <script>
        // Apple-style Scroll Animation Observer
        document.addEventListener("DOMContentLoaded", function() {
            const cards = document.querySelectorAll('.card');

            const observerOptions = {
                root: null,
                rootMargin: '0px 0px -40px 0px',
                threshold: 0.12
            };

            const cardObserver = new IntersectionObserver((entries, observer) => {
                entries.forEach((entry, index) => {
                    if (entry.isIntersecting) {
                        // Каскадно каснење за секоја картичка (Apple stagger effect)
                        setTimeout(() => {
                            entry.target.classList.add('reveal');
                        }, index * 110); 

                        observer.unobserve(entry.target);
                    }
                });
            }, observerOptions);

            cards.forEach(card => cardObserver.observe(card));
        });

        // Двојазичност (MK / EN Toggle Logic)
        let currentLang = 'mk';

        const translations = {
            heroTitle: {
                mk: 'Инженерска прецизност за <span class="highlight-text">станбени и деловни објекти</span>',
                en: 'Engineering precision for <span class="highlight-text">modern structures</span>'
            },
            heroSubtitle: {
                mk: 'Професионално проектирање и изведба на системи за централно греење, климатизација, вентилација, водовод и одводнување.',
                en: 'Professional design and installation of central heating, air conditioning, ventilation, plumbing, and drainage systems.'
            }
        };

        function toggleLanguage() {
            currentLang = currentLang === 'mk' ? 'en' : 'mk';
            
            // Промена на копчето
            document.getElementById('langBtn').innerText = currentLang === 'mk' ? 'EN' : 'MK';

            // Промена на Hero секцијата
            document.getElementById('heroTitle').innerHTML = translations.heroTitle[currentLang];
            document.getElementById('heroSubtitle').innerText = translations.heroSubtitle[currentLang];

            // Промена на сите картички
            const titles = document.querySelectorAll('.card h3');
            const paragraphs = document.querySelectorAll('.card p');

            titles.forEach(el => {
                el.innerText = el.getAttribute(`data-${currentLang}`);
            });

            paragraphs.forEach(el => {
                el.innerText = el.getAttribute(`data-${currentLang}`);
            });
        }
    </script>
